import tkinter as tk
import json
import os


calculation = ""
last_calculation = []


def load_history():
    global last_calculation
    if os.path.exists("history.json"):
        with open("history.json", "r") as file:
            last_calculation = json.load(file)


def save_history():
    with open("history.json", "w") as file:
        json.dump(last_calculation, file)


def add_to_calculation(symbol):
    global calculation
    calculation += str(symbol)
    text_result.delete(1.0, "end")
    text_result.insert(1.0, calculation)


def evaluate_calculation():
    global calculation, last_calculation
    try:
        result = str(eval(calculation))
        text_result.delete(1.0, "end")
        text_result.insert(1.0, result)
        last_calculation.append((calculation, '=', result))
        save_history()  
        calculation = result
    except:
        clear_field()
        text_result.insert(1.0, "Error")


def clear_field():
    global calculation
    calculation = ""
    text_result.delete(1.0, "end")


def history():
    def select_from_history(event):
        selected_index = listbox.curselection()
        if selected_index:
            _, _, result = last_calculation[selected_index[0]]
            text_result.delete(1.0, "end")
            text_result.insert(1.0, result)
            global calculation
            calculation = result

    history_window = tk.Toplevel(root)
    history_window.title("History")
    history_window.geometry("300x200")

    listbox = tk.Listbox(history_window, font=(
        "Arial", 14), width=30, height=10)
    listbox.pack(padx=10, pady=10)

    for calc in last_calculation:
        listbox.insert(tk.END, f"{calc[0]} {calc[1]} {calc[2]}")

    listbox.bind("<<ListboxSelect>>", select_from_history)


root = tk.Tk()
root.geometry("400x420")


load_history()


text_result = tk.Text(root, height=2, width=16, font=("Arial", 24))
text_result.grid(columnspan=5, padx=15, pady=10)
